- 트리거 근거: insight.card.miss: insight 카드 없음: 객체(object_name)별 파지 성공률을 한 번에 조회하고 싶다. 시도 수, 성공 수, 성공률이 필요하다. [corr:7de49816-6b3e-4e0c-9bfc-b112f6fbd1eb]

### 신규 Read Model — Projector

```ts
import { Injectable } from '@nestjs/common';
import { type InferInsertModel } from 'drizzle-orm';
import { PinoLogger } from 'nestjs-pino';
import { ToyDataDto, toyDataSchema } from '@/insert/dto/toy-data.dto';
import { Projector } from '@/projection/projector/projector';
import { EventStoreEventRow } from '@/projection/repository/event-store-reader.repository';
import { DrizzleTx } from '@/shared/database/drizzle.provider';
import { readGripSuccessRate } from '@/shared/database/schema';

type ReadGripSuccessRateInsert = InferInsertModel<typeof readGripSuccessRate>;

@Injectable()
export class GripSuccessRateProjector implements Projector<ReadGripSuccessRateInsert> {
  readonly name: string = "grip-success-rate-projector";

  constructor(private readonly logger: PinoLogger) {
    this.logger.setContext(GripSuccessRateProjector.name);
  }

  map(event: EventStoreEventRow): ReadGripSuccessRateInsert {
    let payload: ToyDataDto;
    try {
      payload = toyDataSchema.parse(event.payload);
    } catch (err) {
      this.logger.error({
        action: LogAction.MAP_FAILED,
        err,
        eventId: event.eventId,
        streamId: event.streamId,
        attemptNum: event.attemptNum,
        globalSeq: event.globalSeq,
      }, "이벤트 매핑(검증) 실패");
      throw err;
    }

    const objectName = payload.objects[0].class_name;
    const gripSucceed = payload.grip_succeed;

    return {
      objectName,
      totalAttempts: 1,
      successfulAttempts: gripSucceed,
      successRate: gripSucceed ? 100 : 0,
    };
  }

  async upsert(tx: DrizzleTx, row: ReadGripSuccessRateInsert): Promise<void> {
    await tx
      .insert(readGripSuccessRate)
      .values(row)
      .onConflictDoUpdate({
        target: [readGripSuccessRate.objectName],
        set: {
          totalAttempts: readGripSuccessRate.totalAttempts + 1,
          successfulAttempts: readGripSuccessRate.successfulAttempts + row.successfulAttempts,
          successRate: ((readGripSuccessRate.successfulAttempts + row.successfulAttempts) * 100) / (readGripSuccessRate.totalAttempts + 1),
        },
      });
  }
}
```
