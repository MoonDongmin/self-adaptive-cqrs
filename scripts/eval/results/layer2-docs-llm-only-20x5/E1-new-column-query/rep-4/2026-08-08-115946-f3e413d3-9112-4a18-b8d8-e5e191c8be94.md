```sql
CREATE TABLE read_sensor_drift_hourly (
  occurred_at timestamptz NOT NULL,
  gripper_temperature doublePrecision,
  conveyor_speed doublePrecision,
  event_count bigint,
  PRIMARY KEY (occurred_at)

```

### 필드


### 투영 매핑 명세 (이벤트 → 컬럼)

> upsert 키: occurred_at · 리플레이: projection_cursor 초기화 시 0 설정. catch-up 전 재투영 시 occurred_at 기준 upsert는 집계(event_count) 누계 또는 overwrite 전제 확인, 멱단성 보장 필요.

| GripAttemptRecorded | gripper_temperature | gripper_temperature | double casting |
| GripAttemptRecorded | conveyor_speed | conveyor_speed | double casting |

파생 컬럼(이벤트 payload 아님):
- `occurred_at` ← extract YYYYMMDD from 2D_image_file_name suffix, format to ISO timestamptz
- `event_count` ← hourly count aggregation per occurred_at bucket

```ts
import { PinoLogger } from 'nestjs-pino';
import type { Projector } from '@/projection/projector/projector';
import type { EventStoreEventRow } from '@/projection/repository/event-store-reader.repository';
import { DrizzleTx } from '@/shared/database/drizzle.provider';
import { readSensorDriftHourly } from '@/shared/database/schema';
import { LogAction, LogContext } from '@/shared/logger/logging-context';

type SensorDriftHourlyProjectorInsert = InferInsertModel<typeof readSensorDriftHourly>;

function toNumberOrNull(value: unknown): number | null {
  if (value === undefined || value === null) {
    return null;
  }
  const parsed = Number(value);
  return Number.isNaN(parsed) ? null : parsed;
}

function toStringOrNull(value: unknown): string | null {
  return value === undefined || value === null ? null : String(value);
}

function toDateOrNull(value: unknown): Date | null {
  if (typeof value !== "string" && typeof value !== "number") {
    return null;
  }
  const parsed = new Date(value);
  return Number.isNaN(parsed.getTime()) ? null : parsed;
}

@Injectable()
export class SensorDriftHourlyProjector implements Projector<SensorDriftHourlyProjectorInsert> {
  readonly name: string = "sensor-drift-hourly-projector";

  constructor(private readonly logger: PinoLogger) {
    this.logger.setContext(SensorDriftHourlyProjector.name);
  }

  map(event: EventStoreEventRow): SensorDriftHourlyProjectorInsert {
    // 결정론 합성 프로젝터 — payload 경로가 자명하지 않은 컬럼은 §2 투영 매핑 명세와 대조해 보정하라.
    const payload = event.payload as Record<string, unknown>;

    this.logger.debug(
      {
        action: LogAction.EVENT_MAPPED,
        [LogContext.PROJECTOR_NAME]: this.name,
        [LogContext.GLOBAL_SEQ]: event.globalSeq,
      },
      "이벤트 매핑",
    );

    return {
      occurredAt: event.occurredAt,
      gripperTemperature: toNumberOrNull(payload["gripper_temperature"]),
      conveyorSpeed: toNumberOrNull(payload["conveyor_speed"]),
      eventCount: toNumberOrNull(payload["event_count"]),
    };
  }

  async upsert(tx: DrizzleTx, row: SensorDriftHourlyProjectorInsert): Promise<void> {
    await tx
      .insert(readSensorDriftHourly)
      .values(row)
      .onConflictDoUpdate({
        target: [readSensorDriftHourly.occurredAt],
        set: {
          gripperTemperature: row.gripperTemperature,
          conveyorSpeed: row.conveyorSpeed,
          eventCount: row.eventCount,
        },
      });
  }
}
```
